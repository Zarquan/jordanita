    podman stop $(podman ps -aq)

    podman rm $(podman ps -aq)

    podman volume rm $(podman volume ls -q)

# -----------------------------------------------------
# Create a shared network, then create three pods on that network.
#[user@desktop]

    podman network create \
        --subnet 172.30.100.0/24 \
        calycopis-network

    podman pod create \
        --name postgres-pod \
        --network calycopis-network \
        --network-alias postgres \
        --publish 5432:5432

    podman pod create \
        --name broker-pod \
        --network calycopis-network \
        --network-alias broker \
        --publish 8082:8082 \
        --publish 3080:3080 \
        --publish 3081:3081


# -----------------------------------------------------
# Start our PostgreSQL database in it's pod.
#[user@desktop]

    podman run \
        --rm \
        --detach \
        --replace \
        --name postgres \
        --pod postgres-pod \
        --env "POSTGRES_DB=calycopis" \
        --env "POSTGRES_USER_FILE=/etc/calycopis/pgusername" \
        --env "POSTGRES_PASSWORD_FILE=/etc/calycopis/pgpassword" \
        --volume "${CONFIG_DIR}:/etc/calycopis:ro,Z" \
        docker.io/library/postgres:latest

# -----------------------------------------------------
# Check we can see our database from the host system.
#[user@desktop]

    pg_isready --host 127.0.0.1

    >   127.0.0.1:5432 - accepting connections


# -----------------------------------------------------
# Run our DSH development environment in it's pod, with writable config.
#[user@desktop]

    # WARNING - this exposes the user's Podman socket to the container.
    # WARNING - exposing the user's Podman socket requires a patch to the SELinux rules.
    # WARNING - this exposes the user's SSH agent to the container.
    # WARNING - exposing the user's SSH agent requires a patch to the SELinux rules.

    source "${HOME:?}/calycopis.env"
    source "${HOME:?}/heliophorus.env"
    podman run \
        --rm \
        --tty \
        --interactive \
        --pod broker-pod \
        --name broker-dev \
        --env "SSH_AUTH_SOCK=/tmp/ssh-sock" \
        --env "CONTAINER_HOST=unix:///run/podman/podman.sock" \
        --volume "${LOG_DIR}:/var/log/calycopis:rw,Z" \
        --volume "${TEST_DIR}:/tmp/calycopis:rw,Z" \
        --volume "${CONFIG_DIR}:/etc/calycopis:rw,Z" \
        --volume "${HOME}/.m2:/root/.m2:rw,Z" \
        --volume "${HOME}/.dsh:/root/.dsh:rw,Z" \
        --volume "${CALYCOPIS_ROOT}:/Calycopis:rw,Z" \
        --volume "${HELIOPHORUS_ROOT}:/Heliophorus:rw,Z" \
        --volume "${SSH_AUTH_SOCK}:/tmp/ssh-sock:rw,Z" \
        --volume "${XDG_RUNTIME_DIR}/podman/podman.sock:/run/podman/podman.sock:rw,Z" \
        localhost/calycopis/deepseek-harness:2026.08.26 \
            bash
