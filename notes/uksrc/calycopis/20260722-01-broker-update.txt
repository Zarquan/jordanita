# Create our platform configuration files.
#[root@calycopis-dev]

    mkdir /etc/calycopis

    cat > /etc/calycopis/database.yaml << EOF
spring:
    datasource:
        url: jdbc:postgresql://postgres:5432/calycopis
        username: $(pwgen 32 1)
        password: $(pwgen 32 1)
        driverClassName: org.postgresql.Driver
        initialize: true
EOF

    cat > /etc/calycopis/admin.yaml << EOF
calycopis:
    admin:
        username: $(pwgen 32 1)
        password: $(pwgen 32 1)
EOF
